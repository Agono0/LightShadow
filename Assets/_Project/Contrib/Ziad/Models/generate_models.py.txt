import math, sys, os
OUT = sys.argv[1]
class Mesh:
    def __init__(s,name): s.name=name; s.v=[]; s.f=[]
    def add(s,verts,faces):
        b=len(s.v); s.v+=verts; s.f+=[tuple(i+b for i in f) for f in faces]
    def vol(s):
        t=0
        for a,b,c in s.f:
            A,B,C=s.v[a],s.v[b],s.v[c]
            t+=(A[0]*(B[1]*C[2]-B[2]*C[1])-A[1]*(B[0]*C[2]-B[2]*C[0])+A[2]*(B[0]*C[1]-B[1]*C[0]))/6
        return t
    def fix(s):
        if s.vol()<0: s.f=[(a,c,b) for a,b,c in s.f]
        return s
import itertools
def closed(f):
    e=set()
    for a,b,c in f:
        for x in((a,b),(b,c),(c,a)):
            if x in e: return False
            e.add(x)
    return all((y,x) in e for x,y in e)
def flipf(f): return [(a,c,b) for a,b,c in f]
def loft(name,axis,st,n=8,c=(0,0),rot=0):
    """st: list of (t, ra, rb, ca, cb) stations along axis; capped both ends."""
    m=Mesh(name); rings=[]
    for (t,ra,rb,ca,cb) in st:
        r=[]
        for i in range(n):
            a=2*math.pi*i/n+rot
            p=(ca+ra*math.cos(a), cb+rb*math.sin(a))
            r.append((p[0],t,p[1]) if axis=='y' else (p[0],p[1],t))
        rings.append(r)
    verts=[p for r in rings for p in r]; faces=[]
    for k in range(len(rings)-1):
        for i in range(n):
            j=(i+1)%n
            a,b,c2,d=k*n+i,k*n+j,(k+1)*n+j,(k+1)*n+i
            faces+= [(a,b,c2),(a,c2,d)]
    side=faces; faces=[]
    for k,top in ((0,False),(len(rings)-1,True)):
        verts.append((sum(p[0] for p in rings[k])/n,sum(p[1] for p in rings[k])/n,sum(p[2] for p in rings[k])/n))
        ci=len(verts)-1
        for i in range(n):
            j=(i+1)%n
            faces.append((ci,k*n+i,k*n+j))
    capb=faces[:n]; capt=faces[n:]
    for fs,fb,ft in itertools.product((0,1),repeat=3):
        cand=(flipf(side) if fs else side)+(flipf(capb) if fb else capb)+(flipf(capt) if ft else capt)
        if closed(cand): faces=cand; break
    else: raise Exception('loft not closable')
    m.add(verts,faces); return m.fix()
def box(name,x0,x1,y0,y1,z0,z1):
    m=Mesh(name)
    v=[(x,y,z) for x in(x0,x1) for y in(y0,y1) for z in(z0,z1)]
    q=[(0,1,3,2),(4,6,7,5),(0,4,5,1),(2,3,7,6),(0,2,6,4),(1,5,7,3)]
    f=[]
    for a,b,c,d in q: f+=[(a,b,c),(a,c,d)]
    m.add(v,f); return m.fix()
def ell(name,cx,cy,cz,rx,ry,rz,seg=8,rings=5):
    m=Mesh(name); v=[(cx,cy+ry,cz)]; f=[]
    for k in range(1,rings):
        th=math.pi*k/rings
        for i in range(seg):
            a=2*math.pi*i/seg
            v.append((cx+rx*math.sin(th)*math.cos(a),cy+ry*math.cos(th),cz+rz*math.sin(th)*math.sin(a)))
    v.append((cx,cy-ry,cz)); bot=len(v)-1
    for i in range(seg):
        j=(i+1)%seg; f.append((0,1+i,1+j))
    for k in range(rings-2):
        for i in range(seg):
            j=(i+1)%seg
            a,b,c,d=1+k*seg+i,1+k*seg+j,1+(k+1)*seg+j,1+(k+1)*seg+i
            f+=[(a,b,c),(a,c,d)]
    base=1+(rings-2)*seg
    f2=[]
    for i in range(seg):
        j=(i+1)%seg; f2.append((bot,base+j,base+i))
    top=f[:seg]; mid=f[seg:]
    for ft,fm,fb in itertools.product((0,1),repeat=3):
        cand=(flipf(top) if ft else top)+(flipf(mid) if fm else mid)+(flipf(f2) if fb else f2)
        if closed(cand): f=cand; break
    else: raise Exception('ell')
    m.add(v,f); return m.fix()
def marker(name,x,y,z,h=0.004):
    return box(name,x-h,x+h,y-h,y+h,z-h,z+h)
def mirror(m,name):
    r=Mesh(name); r.v=[(-x,y,z) for x,y,z in m.v]; r.f=[(a,c,b) for a,b,c in m.f]; return r
def write(fn,mats,objs):
    base=os.path.join(OUT,fn)
    with open(base+'.mtl','w') as f:
        f.write(f"# {fn} material (no textures). Original content.\n")
        f.write(f"newmtl {mats[0]}\nKa 0.02 0.02 0.02\nKd {mats[1]}\nKs 0.03 0.03 0.03\nNs 8\nd 1\nillum 1\n")
    with open(base+'.obj','w') as f:
        f.write(f"# {fn}: units=metres, Y-up, +Z forward, right-handed file convention, scale 1\nmtllib {fn}.mtl\n")
        off=1
        for m in objs:
            f.write(f"o {m.name}\nusemtl {mats[0]}\n")
            for v in m.v: f.write("v %.5f %.5f %.5f\n"%v)
            for a,b,c in m.f: f.write(f"f {a+off} {b+off} {c+off}\n")
            off+=len(m.v)
